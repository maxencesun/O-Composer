# Append this block to OpenOrienteering Mapper's src/CMakeLists.txt after
# Mapper_Common has been declared.
if(EMSCRIPTEN)
  add_executable(MapperConverter
    "${MAPPER_WASM_BRIDGE_SOURCE}"
  )
  target_link_libraries(MapperConverter PRIVATE Mapper_Common Qt5::Widgets)
  target_include_directories(MapperConverter PRIVATE
    "${PROJECT_SOURCE_DIR}/src"
    "${PROJECT_BINARY_DIR}/src"
  )
  set_target_properties(MapperConverter PROPERTIES
    OUTPUT_NAME "mapper-converter"
    SUFFIX ".js"
  )
  target_link_options(MapperConverter PRIVATE
    "SHELL:-s MODULARIZE=1"
    "SHELL:-s EXPORT_ES6=1"
    "SHELL:-s EXPORT_NAME=createMapperConverterModule"
    "SHELL:-s WASM=1"
    "SHELL:-s FULL_ES2=1"
    "SHELL:-s FULL_ES3=1"
    "SHELL:-s USE_WEBGL2=1"
    "SHELL:-s FETCH=1"
    "--bind"
    "SHELL:-s ALLOW_MEMORY_GROWTH=1"
    "SHELL:-s DISABLE_EXCEPTION_CATCHING=0"
    "SHELL:-s NO_EXIT_RUNTIME=1"
    "SHELL:-s TOTAL_MEMORY=134217728"
    "SHELL:-s FORCE_FILESYSTEM=1"
    "SHELL:-s ENVIRONMENT=web"
    "SHELL:-s 'EXPORTED_FUNCTIONS=[\"_main\",\"_malloc\",\"_free\",\"_mapper_convert_ocd\",\"_mapper_output_data\",\"_mapper_output_size\",\"_mapper_warnings_json\",\"_mapper_error_text\",\"_mapper_core_revision\"]'"
    "SHELL:-s 'EXPORTED_RUNTIME_METHODS=[\"UTF8ToString\",\"UTF16ToString\",\"stringToUTF16\"]'"
  )
  if(NOT MAPPER_WASM_PROJ_DATA_DIR OR NOT EXISTS "${MAPPER_WASM_PROJ_DATA_DIR}/proj.db")
    message(FATAL_ERROR
      "MAPPER_WASM_PROJ_DATA_DIR must point to an installed PROJ data directory containing proj.db"
    )
  endif()
  target_link_options(MapperConverter PRIVATE
    "--preload-file"
    "${MAPPER_WASM_PROJ_DATA_DIR}@/share/proj"
  )
endif()
